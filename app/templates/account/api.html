Quick start 
# 1. password: returns a challenge
curl -s -X POST {{ base_url }}/auth/login -H 'Content-Type: application/json' \
  -d '{"username": "{{ current_user.username }}", "password": "..."}'
# 2. your 2FA code: returns a session token
curl -s -X POST {{ base_url }}/auth/verify -H 'Content-Type: application/json' \
  -d '{"challenge": "esc_...", "code": "123456"}'
# 3. call the API with the token (or with a service key created below)
curl -s {{ base_url }}/nodes -H 'Authorization: Bearer est_...'
